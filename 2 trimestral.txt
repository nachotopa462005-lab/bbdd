/*33 join 
SELECT v.titulo AS titulo, v.id_estudio,  e.id_estudio, e.nombre
FROM videojuego AS v
LEFT JOIN estudio AS e ON v.id_estudio = e.id_estudio
ORDER BY titulo ASC; */

/*38 group by 
SELECT d.id_desarrollador, d.nombre, d.id_estudio, SUM(p.horas) as total_horas 
FROM desarrollador as d
JOIN participacion as p on d.id_desarrollador = p.id_desarrollador
GROUP BY d.id_desarrollador, d.nombre 
HAVING total_horas >=500
ORDER BY total_horas DESC;*/

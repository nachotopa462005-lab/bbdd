CREATE OR REPLACE VIEW vw_videojuegos_estudio AS
SELECT
  v.titulo AS videojuego,                   -- Título del videojuego
  e.nombre AS estudio,                      -- Nombre del estudio
  e.pais AS pais_estudio,                   -- País del estudio
  v.pegi,                                   -- Clasificación PEGI
  v.precio_base,                            -- Precio base
  (v.precio_base * 1.21) AS precio_con_iva  -- Precio con IVA
FROM videojuego v
INNER JOIN estudio e                        -- Relación videojuego-estudio
  ON v.id_estudio = e.id_estudio
WHERE v.pegi IS NOT NULL;                   -- Filtra videojuegos con PEGI definido
